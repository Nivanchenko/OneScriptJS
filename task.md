```html

<html>

        имя = "Никита";
        фамилия = "Иванченко";

        сообщить("Привет, ты указал: " + имя + ", " + фамилия);

            сообщить("Добро пожаловать, " + имя + " твой возраст " + возраст);
        конецесли;

    </one-script>
</body>
</html>
```
